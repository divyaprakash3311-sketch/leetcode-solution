LEETCODE-1295

class Solution {
    public int findNumbers(int[] nums) {
        int count1=0;
       
        for(int i=0;i<nums.length;i++){
             int count=0;
            while(nums[i]>0){
                
                nums[i]=nums[i]/10;
                count++;
            }
            if(count%2==0){
               count1++;
            }
        }
        return count1;
    }
}

LEETCODE-704

class Solution {
    public int search(int[] nums, int target) {
       int left = 0;
    int right = nums.length-1;
    while (left <= right) {
        int mid = (left + right)/2;
        if (target < nums[mid]) {
            right = mid - 1;
        } else if (target>nums[mid]) {
            left = mid+1; 
        }
        else{
            return mid;
        }
    }
    return -1;
}
}

LEETCODE-66

class Solution {
    public int[] plusOne(int[] digits) {
        for (int i = digits.length - 1; i >= 0; i--) {
            if (digits[i] + 1 != 10) {
                digits[i] += 1;
                return digits;
            }
            digits[i] = 0;
        }
        
        int[] b = new int[digits.length + 1];
        b[0] = 1;
        return b;        
    }
}
